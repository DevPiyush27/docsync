-- 1. Create oops-assignments bucket INSERT INTO storage.buckets (id, name, public) VALUES ('oops-assignments', 'oops-assignments', true) ON CONFLICT (id) DO UPDATE SET public = true; -- 2. Allow anyone to list and download files CREATE POLICY "Public Access" ON storage.objects FOR SELECT TO public USING (bucket_id = 'oops-assignments'); -- 3. Allow anyone to upload files CREATE POLICY "Public Upload" ON storage.objects FOR INSERT TO public WITH CHECK (bucket_id = 'oops-assignments'); -- 4. Allow overwrite / updates CREATE POLICY "Public Update" ON storage.objects FOR UPDATE TO public USING (bucket_id = 'oops-assignments');
